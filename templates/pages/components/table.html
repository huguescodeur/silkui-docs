Showing – of rows 

{% verbatim %}{% load silkui_tags %}

<div x-data="{
  page: 1, perPage: 5,
  rows: [...],
  get totalPages() { return Math.ceil(this.rows.length / this.perPage) },
  get paginated() { return this.rows.slice((this.page-1)*this.perPage, this.page*this.perPage) },
  get from() { return (this.page-1)*this.perPage + 1 },
  get to() { return Math.min(this.page*this.perPage, this.rows.length) }
}">
  {% table %}
    {% table_header %}
      {% table_row %}
        {% table_head %}Invoice{% endtable_head %}
        {% table_head %}Customer{% endtable_head %}
        {% table_head class="text-right" %}Amount{% endtable_head %}
      {% endtable_row %}
    {% endtable_header %}
    {% table_body %}
      <template x-for="row in paginated" :key="row.id">
        <tr class="border-b transition-colors hover:bg-muted/50">
          <td class="p-4 font-medium" x-text="row.id"></td>
          <td class="p-4" x-text="row.name"></td>
          <td class="p-4 text-right" x-text="row.amount"></td>
        </tr>
      </template>
    {% endtable_body %}
  {% endtable %}

  <div class="flex items-center justify-between pt-4">
    <p class="text-sm text-muted-foreground">
      Showing <span x-text="from"></span>–<span x-text="to"></span> of <span x-text="rows.length"></span>
    </p>
    <div class="flex gap-1">
      <button @click="page--" :disabled="page === 1" class="...">Previous</button>
      <template x-for="p in totalPages" :key="p">
        <button @click="page = p" :class="page===p ? 'bg-primary text-primary-foreground':'hover:bg-accent'" class="...">
          <span x-text="p"></span>
        </button>
      </template>
      <button @click="page++" :disabled="page === totalPages" class="...">Next</button>
    </div>
  </div>

{% verbatim %}{% load silkui_tags %}

<div x-data="{
  sortKey: 'name', sortDir: 'asc',
  rows: [...],
  sort(key) {
    if (this.sortKey === key) { this.sortDir = this.sortDir === 'asc' ? 'desc' : 'asc' }
    else { this.sortKey = key; this.sortDir = 'asc' }
  },
  get sorted() {
    return [...this.rows].sort((a, b) => {
      let av = a[this.sortKey], bv = b[this.sortKey]
      if (typeof av === 'number') return this.sortDir === 'asc' ? av - bv : bv - av
      return this.sortDir === 'asc' ? av.localeCompare(bv) : bv.localeCompare(av)
    })
  }
}">
  {% table %}
    {% table_header %}
      {% table_row %}
        <th @click="sort('name')" class="h-12 px-4 text-left align-middle font-medium text-muted-foreground cursor-pointer hover:text-foreground">
          Name
        </th>
        <th @click="sort('score')" class="h-12 px-4 text-left align-middle font-medium text-muted-foreground cursor-pointer hover:text-foreground">
          Score
        </th>
      {% endtable_row %}
    {% endtable_header %}
    {% table_body %}
      <template x-for="row in sorted" :key="row.name">
        <tr class="border-b transition-colors hover:bg-muted/50">
          <td class="p-4 align-middle font-medium" x-text="row.name"></td>
          <td class="p-4 align-middle" x-text="row.score"></td>
        </tr>
      </template>
    {% endtable_body %}
  {% endtable %}
</div>{% endverbatim %}
